## 2. Architecture / proposed approach

多入口静态站：每个 URL 一个 `*.html` 入口（沿用已验收的 `/ai` 机制），共享 `src/shell/` 与 `src/content/`，页面本身只是区块组合。Express 增加 5 个 `/api/*` 路由，全部先经 `server/tools/*` 的剥价层，再经 `server/leads.ts` 的留资令牌闸门；引擎仍只由 `deploy/osrm-engine/server.py` 窄暴露，白名单从 1 个业务端点扩到 3 个。AI 助手不新增协议，只在 `server/agentChat.ts` 里把已有工具从 1 个扩到 3 个，复用同一条 SSE 与同一份剥价函数。

/** 数值 env 归一：非有限/≤0 回落默认并 console.warn，**非法值必须留痕**。 */
export function numEnv(name: string, fallback: number, opts: { min?: number; max?: number } = {}): number {
  const raw = process.env[name];
  if (raw === undefined || raw.trim() === '') return fallback;
  const v = Number(raw);
  const min = opts.min ?? Number.NEGATIVE_INFINITY;
  const max = opts.max ?? Number.POSITIVE_INFINITY;
  if (!Number.isFinite(v) || v <= 0 || v < min || v > max) {
    console.warn(`[env] ${name}=${JSON.stringify(raw)} 非法，回落默认 ${fallback}`);
    return fallback;
  }
  return v;
}

export function strEnv(name: string, fallback = ''): string {
  const v = process.env[name];
  return v === undefined ? fallback : v.trim();
}

export const v7 = {
  borderMarkup: () => (strEnv('BORDER_MARKUP') === '' ? null : numEnv('BORDER_MARKUP', 1.5, { min: 1, max: 10 })),
  borderRangePct: () => numEnv('BORDER_RANGE_PCT', 0.1, { min: 0.01, max: 0.5 }),
  leadTtlMin: () => numEnv('LEAD_TTL_MIN', 30, { min: 1, max: 1440 }),
  leadRatePerHour: () => numEnv('LEAD_RATE_PER_HOUR', 5, { min: 1, max: 100 }),
  detailSecret: () => strEnv('DETAIL_TOKEN_SECRET', 'dev-only-secret'),
  leadsDir: () => strEnv('LEADS_DIR', './data'),
  leadsWebhook: () => strEnv('LEADS_WEBHOOK_URL'),
};

const LANGS = ['zh', 'vi', 'en'];

function load(lang, f) {

function keys(obj, pre = '') {

    v && typeof v === 'object' && !Array.isArray(v) ? keys(v, `${pre}${k}.`) : [`${pre}${k}`],
